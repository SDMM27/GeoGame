# Flag Game Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement a "Trouve le Drapeau" mini-game where users see a flag SVG and must type the country name in French.

**Architecture:** Offline SVG flags via `country-flag-icons` npm package (keyed by ISO A2). JSON enriched with `ISO_A2` codes via a Python script. New hook `useFlagGameState` + page `FlagGamePage`, wired into existing router and homepage.

**Tech Stack:** React, Vitest, country-flag-icons, pycountry (Python, one-shot script), Firebase Firestore (existing)

```python
import json
import pycountry

INPUT = 'src/data/countriesWithDifficulty_FR.json'

with open(INPUT, 'r', encoding='utf-8') as f:
    data = json.load(f)

not_found = []
for feature in data['features']:
    props = feature['properties']
    iso_a3 = props.get('ISO_A3', '')
    iso_a2 = None
    if iso_a3 and iso_a3 != '-99':
        country = pycountry.countries.get(alpha_3=iso_a3)
        if country:
            iso_a2 = country.alpha_2
        else:
            not_found.append(iso_a3)
    props['ISO_A2'] = iso_a2
    props['aliases'] = []

with open(INPUT, 'w', encoding='utf-8') as f:
    json.dump(data, f, ensure_ascii=False, separators=(',', ':'))

print(f"Done. {len(not_found)} codes not resolved: {not_found}")
```

Create `src/utils/normalizeAnswer.js`:

```js
export function normalize(str) {
  return str
    .toLowerCase()
    .normalize('NFD').replace(/[\u0300-\u036f]/g, '')
    .replace(/[-']/g, ' ')
    .replace(/\s+/g, ' ')
    .trim()
}

export function checkAnswer(input, target, aliases = []) {
  const normalizedInput = normalize(input)
  const candidates = [target, ...aliases].map(normalize)
  return candidates.includes(normalizedInput)
}
```

- [ ] **Step 1: Create the hook**

Create `src/hooks/useFlagGameState.js`:

```js
import { useState, useEffect, useRef, useCallback } from 'react'
import { useNavigate } from 'react-router-dom'
import { toast } from 'react-toastify'
import countries from '../data/countriesWithDifficulty_FR.json'
import { formatTime } from '../utils/formatTime'
import { checkAnswer } from '../utils/normalizeAnswer'

const DIFFICULTIES = ['Facile', 'Moyen', 'Difficile', 'Impossible', 'Hardcore']

const SCORE_TABLE = {
  Facile:     [3, 2, 1],
  Moyen:      [5, 3, 2],
  Difficile:  [7, 5, 3],
  Impossible: [9, 7, 5],
}

function getAvailableDifficulties(difficulty) {
  if (difficulty === 'Hardcore') return ['Impossible']
  return DIFFICULTIES.slice(0, DIFFICULTIES.indexOf(difficulty) + 1)
}

export function getTotalFlagCountries(difficulty) {
  const available = getAvailableDifficulties(difficulty)
  return countries.features.filter(c =>
    available.includes(c.properties.Difficulty) && c.properties.ISO_A2
  ).length
}

let alreadyChosenCountries = []

export default function useFlagGameState(difficulty, isTraining) {
  const navigate = useNavigate()

  const [targetCountry, setTargetCountry] = useState(null)
  const targetCountryRef = useRef(null)
  const [score, setScore] = useState(0)
  const [gameStarted, setGameStarted] = useState(false)
  const gameStartedRef = useRef(false)
  const [consecutiveErrors, setConsecutiveErrors] = useState(0)
  const consecutiveErrorsRef = useRef(0)
  const [inputValue, setInputValue] = useState('')
  const [revealedName, setRevealedName] = useState(null)
  const [stats, setStats] = useState({ found: 0, skipped: 0, total: 0 })
  const [timer, setTimer] = useState(0)
  const timerRef = useRef(0)
  const startTimeRef = useRef(null)
  const [clickOnSeeCountry, setClickOnSeeCountry] = useState(false)
  const clickOnSeeCountryRef = useRef(false)
  const gameOverToastShownRef = useRef(false)

  useEffect(() => {
    if (!difficulty) navigate('/')
  }, [difficulty, navigate])

  const getRandomCountry = useCallback((diff) => {
    const available = getAvailableDifficulties(diff)
    const filtered = countries.features.filter(c =>
      available.includes(c.properties.Difficulty) &&
      c.properties.ISO_A2 &&
      !alreadyChosenCountries.includes(c.properties.ADMIN)
    )
    if (filtered.length === 0) return null
    return filtered[Math.floor(Math.random() * filtered.length)]
  }, [])

  useEffect(() => {
    if (!difficulty) return
    alreadyChosenCountries = []
    const total = getTotalFlagCountries(difficulty)
    setStats({ found: 0, skipped: 0, total })
    const first = getRandomCountry(difficulty)
    if (first) {
      alreadyChosenCountries.push(first.properties.ADMIN)
      setTargetCountry(first)
      gameStartedRef.current = true
      setGameStarted(true)
    }
  }, []) // eslint-disable-line react-hooks/exhaustive-deps

  useEffect(() => { targetCountryRef.current = targetCountry }, [targetCountry])
  useEffect(() => { consecutiveErrorsRef.current = consecutiveErrors }, [consecutiveErrors])
  useEffect(() => { clickOnSeeCountryRef.current = clickOnSeeCountry }, [clickOnSeeCountry])

  const allProcessed = stats.total > 0 && (stats.found + stats.skipped >= stats.total)

  useEffect(() => {
    if (!gameStarted || allProcessed) {
      if (allProcessed && startTimeRef.current) {
        timerRef.current = performance.now() - startTimeRef.current
      }
      return
    }
    startTimeRef.current = performance.now() - (timerRef.current || 0)
    const id = setInterval(() => {
      const currentTime = performance.now() - startTimeRef.current
      setTimer(currentTime)
      timerRef.current = currentTime
    }, 100)
    return () => clearInterval(id)
  }, [gameStarted, allProcessed])

  useEffect(() => {
    if (!allProcessed || !gameStarted || gameOverToastShownRef.current) return
    gameOverToastShownRef.current = true
    if (isTraining) {
      toast.success('Bravo ! Vous avez trouvé tous les drapeaux !', {
        containerId: 'centerToast',
      })
    }
  }, [allProcessed, gameStarted, isTraining])

  const incrementScore = (diff, errorCount) => {
    const row = SCORE_TABLE[diff]
    if (!row) return
    const adjustedErrors = Math.max(0, errorCount - 1)
    const gain = row[Math.min(adjustedErrors, 2)]
    setScore(prev => prev + gain)
  }

  const changeCountry = useCallback(() => {
    const next = getRandomCountry(difficulty)
    if (next) {
      alreadyChosenCountries.push(next.properties.ADMIN)
      setTargetCountry(next)
    }
    setInputValue('')
    setRevealedName(null)
    setConsecutiveErrors(0)
    consecutiveErrorsRef.current = 0
    setClickOnSeeCountry(false)
    clickOnSeeCountryRef.current = false
  }, [difficulty, getRandomCountry])

  // Auto-advance after reveal (3 errors or "voir")
  useEffect(() => {
    if (!revealedName) return
    const id = setTimeout(() => {
      setStats(prev => ({ ...prev, skipped: prev.skipped + 1 }))
      changeCountry()
    }, 2000)
    return () => clearTimeout(id)
  }, [revealedName, changeCountry])

  const handleSubmit = () => {
    if (!gameStartedRef.current || !targetCountryRef.current || revealedName) return
    const props = targetCountryRef.current.properties
    const isCorrect = checkAnswer(inputValue, props.ADMIN, props.aliases ?? [])

    if (isCorrect) {
      const wasSkipped = clickOnSeeCountryRef.current
      if (!wasSkipped) {
        toast.success('Bravo ! ' + props.ADMIN + ' !', { containerId: 'topRightToast' })
        incrementScore(props.Difficulty, consecutiveErrorsRef.current)
      }
      setStats(prev =>
        wasSkipped
          ? { ...prev, skipped: prev.skipped + 1 }
          : { ...prev, found: prev.found + 1 }
      )
      changeCountry()
    } else {
      const newErrors = consecutiveErrorsRef.current + 1
      setConsecutiveErrors(newErrors)
      consecutiveErrorsRef.current = newErrors
      if (newErrors >= 3) {
        toast.error('Dommage ! C\'est ' + props.ADMIN, { containerId: 'topRightToast' })
        setRevealedName(props.ADMIN)
      } else {
        toast.error('Dommage !', { containerId: 'topRightToast' })
      }
      setInputValue('')
    }
  }

  const seeCountry = () => {
    if (revealedName) return
    setClickOnSeeCountry(true)
    clickOnSeeCountryRef.current = true
    setRevealedName(targetCountryRef.current?.properties?.ADMIN ?? null)
  }

  const restartGame = () => {
    setGameStarted(false)
    gameStartedRef.current = false
    alreadyChosenCountries = []
    setScore(0)
    setConsecutiveErrors(0)
    setInputValue('')
    setRevealedName(null)
    setClickOnSeeCountry(false)
    clickOnSeeCountryRef.current = false
    gameOverToastShownRef.current = false
    startTimeRef.current = null
    timerRef.current = 0
    setTimer(0)
    setTimeout(() => {
      const total = getTotalFlagCountries(difficulty)
      setStats({ found: 0, skipped: 0, total })
      const next = getRandomCountry(difficulty)
      if (next) {
        alreadyChosenCountries.push(next.properties.ADMIN)
        setTargetCountry(next)
        setGameStarted(true)
        gameStartedRef.current = true
      }
    }, 0)
  }

  const goHome = () => {
    alreadyChosenCountries = []
    navigate('/')
  }

  const isGameOver = allProcessed && !isTraining
  const displayTimer = isTraining ? '∞' : formatTime(timer)

  return {
    targetCountry,
    score,
    timer,
    stats,
    isGameOver,
    displayTimer,
    inputValue,
    setInputValue,
    handleSubmit,
    revealedName,
    seeCountry,
    restartGame,
    goHome,
  }
}
```

```jsx
          : <div className="flag-placeholder" />
        }

        {revealedName && (
          <div className="flag-revealed-name">{revealedName}</div>
        )}

        <div className="flag-input-row">
          <input
            ref={inputRef}
            className="flag-input"
            type="text"
            placeholder="Tape le nom du pays..."
            value={inputValue}
            onChange={e => setInputValue(e.target.value)}
            onKeyDown={onKeyDown}
            disabled={!!revealedName || isGameOver}
          />
          <button
            className="flag-submit-btn"
            onClick={handleSubmit}
            disabled={!!revealedName || isGameOver || !inputValue.trim()}
          >
            →
          </button>
        </div>
      </div>

      {/* Game over overlay */}
      {isGameOver && (
        <div className="game-over-overlay">
          <div className="game-over-content">
            <Form finalScore={score} difficulty={difficulty} elapsedTime={timer} gameId="flags" />
            <div className="game-over-actions">
              <button className="hud-btn hud-btn--restart" onClick={restartGame}>↺ Recommencer</button>
              <button className="hud-btn hud-btn--home" onClick={goHome}>← Accueil</button>
            </div>
          </div>
        </div>
      )}

      <ToastContainer
        enableMultiContainer
        containerId="centerToast"
        position="bottom-center"
        autoClose={3000}
        theme="dark"
        hideProgressBar
      />
      <ToastContainer
        enableMultiContainer
        containerId="topRightToast"
        position="top-right"
        autoClose={1500}
        theme="dark"
        hideProgressBar
      />
    </div>
  )
}
```
